    16	function hexToRgb(h) {
    17	  if (h[0] !== "#") { // "rgb(r,g,b)" oder "rgba(...)"
    18	    const m = h.match(/[\d.]+/g); return m ? [+m[0], +m[1], +m[2]] : [0, 0, 0];
    19	  }
    20	  const n = parseInt(h.slice(1), 16); return [(n >> 16) & 255, (n >> 8) & 255, n & 255];
    21	}
    22	function lerpColor(a, b, t) {
    23	  const A = hexToRgb(a), B = hexToRgb(b);
    24	  return `rgb(${Math.round(lerp(A[0], B[0], t))},${Math.round(lerp(A[1], B[1], t))},${Math.round(lerp(A[2], B[2], t))})`;
    25	}
    26	function getPalette(t, raw = false) {
    27	  let pal = SKY_KEYS[0];
    28	  for (let i = 0; i < SKY_KEYS.length - 1; i++) {
    29	    const a = SKY_KEYS[i], b = SKY_KEYS[i + 1];
    30	    if (t >= a.t && t <= b.t) {
    31	      const k = (t - a.t) / (b.t - a.t);
    32	      pal = { top: lerpColor(a.top, b.top, k), bot: lerpColor(a.bot, b.bot, k), wTop: lerpColor(a.wTop, b.wTop, k), wBot: lerpColor(a.wBot, b.wBot, k), light: lerp(a.light, b.light, k) };
    33	      break;
    34	    }
    35	  }
    36	  // Ort färbt das Wasser und kann es abdunkeln (Tiefsee)
    37	  if (!raw && typeof getLocation === "function" && typeof save !== "undefined") {
    38	    const loc = getLocation();
    39	    if (loc.id !== "see" || loc.dark) {
    40	      const night = 1 - pal.light;
    41	      pal = Object.assign({}, pal, {
    42	        wTop: lerpColor(loc.water[0], "#06141f", night * 0.85),
    43	        wBot: lerpColor(loc.water[1], "#020810", night * 0.85),
    44	        light: pal.light * (1 - loc.dark)
    45	      });
    46	      if (loc.dark) { pal.top = lerpColor(pal.top, "#0a0f1e", loc.dark); pal.bot = lerpColor(pal.bot, "#1a2438", loc.dark); }
    47	    }
    48	    // Regenwetter zieht den Himmel zu und graut das Wasser ein (#104).
    49	    // Nur die FARBEN — `light` bleibt, sonst kämen Nachtfische am Regentag.
    50	    const g = typeof weatherGloom === "function" ? weatherGloom() : 0;
    51	    if (g > 0.01 && !loc.dark) {
    52	      pal = Object.assign({}, pal, {
    53	        top: lerpColor(pal.top, "#5a6673", g * 0.55),
    54	        bot: lerpColor(pal.bot, "#7d8894", g * 0.5),
    55	        wTop: lerpColor(pal.wTop, "#4a5a66", g * 0.35),
    56	        wBot: lerpColor(pal.wBot, "#26313a", g * 0.3)
    57	      });
    58	    }
    59	  }
    60	  return pal;
    61	}

    64	function celestialPos(p) {
    65	  return { x: canvas.width * (0.08 + 0.84 * p), y: horizonY + canvas.height * 0.03 - Math.sin(p * Math.PI) * canvas.height * 0.33 };
    66	}
    67	function updateCelestials() {
    68	  const p = (dayTime - 0.12) / 0.76;
    69	  const s = celestialPos(clamp(p, -0.1, 1.1));
    70	  sunX = s.x; sunY = s.y;
    71	  sunRadius = Math.min(canvas.width * 0.08, canvas.height * 0.11);
    72	  sunVisible = p > -0.05 && p < 1.05;
    73	  const q = ((dayTime + 0.5) % 1 - 0.12) / 0.76;
    74	  const m = celestialPos(clamp(q, -0.1, 1.1));
    75	  moonX = m.x; moonY = m.y; moonVisible = q > -0.05 && q < 1.05;
    76	}

    94	  const waterGradient = ctx.createLinearGradient(0, horizonY, 0, canvas.height);
    95	  waterGradient.addColorStop(0, pal.wTop);

    97	  ctx.fillStyle = waterGradient;
    98	  ctx.fillRect(0, horizonY - 12, canvas.width, canvas.height - horizonY + 12);

   101	function drawStars() {
   102	  const light = getPalette(dayTime, true).light;
   103	  if (light > 0.6) return;
   104	  const a = clamp((0.6 - light) / 0.5, 0, 1);
   105	  ctx.save();
   106	  ctx.globalAlpha = a;
   107	  ctx.fillStyle = "#ffffff";
   108	  for (let i = 0; i < 90; i++) {
   109	    // deterministische Pseudo-Zufallspositionen
   110	    const sx = ((i * 7919) % 1000) / 1000 * canvas.width;
   111	    const sy = ((i * 104729) % 1000) / 1000 * horizonY * 0.9;
   112	    const tw = 0.5 + 0.5 * Math.sin(time * 2 + i);
   113	    ctx.globalAlpha = a * (0.4 + 0.6 * tw);
   114	    ctx.beginPath(); ctx.arc(sx, sy, i % 5 === 0 ? 1.8 : 1.1, 0, Math.PI * 2); ctx.fill();
   115	  }
   116	  ctx.restore();
   117	}

   123	function drawSun() {
   124	  if (!sunVisible) return;
   125	  clipToSky();
   126	  // Bei Regen steht die Sonne hinter der Wolkendecke (#104)
   127	  const g = typeof weatherGloom === "function" ? weatherGloom() : 0;

   129	  drawSunBody();

   132	function drawSunBody() {
   133	  // Tief am Horizont wird die Sonne optisch größer, wärmer und weicher —
   134	  // die Morgen-/Abendsonne ist eine andere als die Mittagssonne.
   135	  const low = clamp(1 - (horizonY - sunY) / (canvas.height * 0.3), 0, 1);
   136	  const r = sunRadius * (1 + low * 0.3);
   137	  const core = lerpColor("#fff6c9", "#ffe9b0", low);
   138	  const mid = lerpColor("#ffe066", "#ff9d4d", low);
   139	  const rim = lerpColor("#ffc94d", "#ff7a30", low);
   140	
   141	  // sehr weiter, weicher Schein (zwei Lagen statt einer harten)
   142	  const glow2 = ctx.createRadialGradient(sunX, sunY, 0, sunX, sunY, r * 3.4);
   143	  glow2.addColorStop(0, `rgba(255,220,150,${0.22 + low * 0.1})`);
   144	  glow2.addColorStop(0.5, `rgba(255,200,130,${0.08 + low * 0.06})`);
   145	  glow2.addColorStop(1, "rgba(255,200,130,0)");
   146	  ctx.fillStyle = glow2;
   147	  ctx.beginPath(); ctx.arc(sunX, sunY, r * 3.4, 0, Math.PI * 2); ctx.fill();
   148	  const glow1 = ctx.createRadialGradient(sunX, sunY, r * 0.5, sunX, sunY, r * 1.7);
   149	  glow1.addColorStop(0, "rgba(255,240,170,0.35)");
   150	  glow1.addColorStop(1, "rgba(255,240,170,0)");
   151	  ctx.fillStyle = glow1;
   152	  ctx.beginPath(); ctx.arc(sunX, sunY, r * 1.7, 0, Math.PI * 2); ctx.fill();
   153	
   154	  const sunGradient = ctx.createRadialGradient(sunX - r * 0.3, sunY - r * 0.3, r * 0.1, sunX, sunY, r);
   155	  sunGradient.addColorStop(0, core);
   156	  sunGradient.addColorStop(0.6, mid);
   157	  sunGradient.addColorStop(1, rim);
   158	  ctx.fillStyle = sunGradient;
   159	  ctx.beginPath(); ctx.arc(sunX, sunY, r, 0, Math.PI * 2); ctx.fill();
   160	}

   162	function drawMoon() {
   163	  const light = getPalette(dayTime, true).light;
   164	  if (!moonVisible || light > 0.75) return;
   165	  const r = sunRadius * 0.6;
   166	  clipToSky();
   167	  ctx.globalAlpha = clamp((0.75 - light) / 0.4, 0, 1);
   168	  // Der Mond ist jede Nacht ein `shadowBlur` je Bild (ein Gauss-Durchlauf
   169	  // ueber ~200 px). Er aendert an sich selbst aber NIE etwas — nur Ort und
   170	  // Deckkraft. Also einmal gebacken (Scheibe MIT ihrem Halo plus Krater)
   171	  // und als Stempel an die gebrochene Position kopiert; die Deckkraft kommt
   172	  // wie vorher ueber globalAlpha. Mit `Blech.an = false` zeichnet der
   173	  // Rueckfallweg des Stempels exakt den alten Ablauf.
   174	  Blech.stempel("mond", [Math.round(r * 4)], moonX, moonY, r * 1.5 + 6, r * 1.5 + 6, () => {
   175	    ctx.shadowColor = "rgba(200,210,255,0.6)"; ctx.shadowBlur = r;
   176	    ctx.fillStyle = "#e9edff";
   177	    ctx.beginPath(); ctx.arc(0, 0, r, 0, Math.PI * 2); ctx.fill();
   178	    ctx.shadowBlur = 0;
   179	    ctx.fillStyle = "rgba(160,170,210,0.35)";
   180	    ctx.beginPath(); ctx.arc(-r * 0.3, -r * 0.2, r * 0.22, 0, Math.PI * 2); ctx.fill();
   181	    ctx.beginPath(); ctx.arc(r * 0.35, r * 0.3, r * 0.15, 0, Math.PI * 2); ctx.fill();
   182	    ctx.beginPath(); ctx.arc(r * 0.1, -r * 0.5, r * 0.1, 0, Math.PI * 2); ctx.fill();
   183	  });
   184	  ctx.restore();
   185	}

   187	function drawClouds() {
   188	  const light = getPalette(dayTime).light;
   189	  // Regen: die Wolken werden grau, schwer und dichter — der Himmel zieht
   190	  // sichtbar zu, statt dass es aus heiterem Blau schuettet (#104)
   191	  const g = typeof weatherGloom === "function" ? weatherGloom() : 0;
   192	  const col = g > 0.01 ? lerpColor("#ffffff", "#66707c", g * 0.75) : "#ffffff";
   193	  const rgbw = hexToRgb(col);
   194	  ctx.save();
   195	  // HIER STAND EIN STEMPEL (blech.js). Zurueckgerollt: der Schluessel fasste
   196	  // Wolken nach halben CSS-Pixeln zusammen, die Signatur trug aber die exakte
   197	  // Geraetepixel-Groesse — zwei Wolken derselben Gruppe warfen sich also
   198	  // gegenseitig heraus. Und ueber 192 Geraetepixel Kante (auf einem 3x-Schirm
   199	  // fast jede Wolke) hat der Stempel ohnehin nur noch direkt gezeichnet, mit
   200	  // dem Umweg ueber Blech obendrauf.
   201	  const aOben = 0.25 + 0.6 * light + g * 0.2;
   202	  const aUnten = 0.07 * light + g * 0.32;
   203	  for (const c of clouds) {
   204	    const x = c.x * canvas.width, y = c.y * horizonY, s = c.s * canvas.width * 0.05 * (1 + g * 0.35);
   205	    ctx.fillStyle = `rgba(${rgbw[0]},${rgbw[1]},${rgbw[2]},${aOben})`;
   206	    ctx.beginPath();
   207	    ctx.ellipse(x, y, s * 1.6, s * 0.55, 0, 0, Math.PI * 2);
   208	    ctx.ellipse(x - s * 0.7, y + s * 0.1, s * 0.9, s * 0.45, 0, 0, Math.PI * 2);
   209	    ctx.ellipse(x + s * 0.7, y + s * 0.05, s, s * 0.5, 0, 0, Math.PI * 2);
   210	    ctx.ellipse(x + s * 0.1, y - s * 0.3, s * 0.9, s * 0.55, 0, 0, Math.PI * 2);
   211	    ctx.fill();
   212	    // Unterseite: Wolken haben immer eine leichte Schattenseite unten —
   213	    // bei Regen wird sie schwer und dunkel
   214	    ctx.fillStyle = `rgba(90,105,120,${aUnten})`;
   215	    ctx.beginPath(); ctx.ellipse(x, y + s * 0.3, s * 1.4, s * 0.3, 0, 0, Math.PI * 2); ctx.fill();
   216	  }
   217	  ctx.restore();
   218	}

   220	function drawBirds() {
   221	  if (getPalette(dayTime).light < 0.5) return;
   222	  ctx.save();
   223	  ctx.strokeStyle = "rgba(30,30,40,0.8)";
   224	  ctx.lineWidth = 2;
   225	  // Ein Vogel sind zwei Kurven. Er stand kurz auf dem Blech (33 winzige
   226	  // Motive fuer den Fluegelschlag); zurueckgerollt mit dem Rest des
   227	  // Angelplatzes — zwei Kurven sind billiger als der Weg ueber den Cache.
   228	  for (const b of birds) {
   229	    const x = b.x * canvas.width, y = b.y * horizonY;
   230	    const w = 9, flap = Math.sin(b.phase) * 5;
   231	    ctx.beginPath();
   232	    ctx.moveTo(x - w, y - flap); ctx.quadraticCurveTo(x - w / 2, y + 2, x, y);
   233	    ctx.quadraticCurveTo(x + w / 2, y + 2, x + w, y - flap);
   234	    ctx.stroke();
   235	  }
   236	  ctx.restore();
   237	}

   240	function drawWater() {
   241	  const pal = getPalette(dayTime);
   242	  const layers = [
   243	    { off: 18, alpha: 0.18, phase: 1.7, color: pal.wBot },
   244	    { off: 8,  alpha: 0.25, phase: 0.8, color: pal.wTop },
   245	    { off: 0,  alpha: 0.55, phase: 0,   color: pal.wTop }
   246	  ];
   247	  for (const L of layers) {
   248	    ctx.save();
   249	    ctx.globalAlpha = L.alpha;
   250	    ctx.fillStyle = L.color;
   251	    ctx.beginPath();
   252	    ctx.moveTo(0, horizonY + L.off);
   253	    for (let x = 0; x <= canvas.width; x += 5) ctx.lineTo(x, getWave(x, L.phase) + L.off);
   254	    ctx.lineTo(canvas.width, horizonY + 60); ctx.lineTo(0, horizonY + 60);
   255	    ctx.closePath(); ctx.fill();
   256	    ctx.restore();
   257	  }
   258	  // Schaumlinie
   259	  ctx.save();
   260	  ctx.strokeStyle = `rgba(255,255,255,${0.15 + 0.25 * pal.light})`;
   261	  ctx.lineWidth = 1.5;
   262	  ctx.beginPath();
   263	  for (let x = 0; x <= canvas.width; x += 5) { const y = getWave(x); x === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y); }
   264	  ctx.stroke();
   265	  ctx.restore();
   266	
   267	  // Sonnen-/Mondglitzern
   268	  const src = sunVisible && pal.light > 0.3 ? { x: sunX, a: 0.35 * pal.light } : (moonVisible ? { x: moonX, a: 0.25 } : null);
   269	  if (src) {
   270	    ctx.save();
   271	    for (let y = horizonY + 12; y < horizonY + canvas.height * 0.3; y += 7) {
   272	      const spread = (y - horizonY) * 0.9 + 10;
   273	      const n = 3;
   274	      for (let k = 0; k < n; k++) {
   275	        const seed = Math.sin(y * 0.37 + k * 2.1 + time * 1.7) * Math.cos(y * 0.11 + time * 0.9 + k);
   276	        if (seed < 0.35) continue;
   277	        const gx = src.x + Math.sin(y * 1.3 + k * 4.7) * spread;
   278	        ctx.globalAlpha = src.a * (seed - 0.35) / 0.65;
   279	        ctx.fillStyle = "#ffffff";
   280	        ctx.fillRect(gx - 8 - k * 3, y, 16 + k * 6, 2);
   281	      }
   282	    }
   283	    ctx.restore();
   284	  }
   285	}

   334	function _drawAngler() {
   335	  const s = dockHeight;
   336	  const x = rodBaseX + s * 1.2, y = dockY;
   337	  const o = getOutfit();
   338	  // Licht: die Figur haengt am Tageslicht wie die Szene — mittags klar
   339	  // modelliert, nachts flach und kuehl. Kein fester Studio-Look mehr.
   340	  const dl = daylight();
   341	  const dark = c => shadeColor(c, -(0.24 + 0.1 * (1 - dl)));
   342	  const light = c => shadeColor(c, 0.12 + 0.14 * dl);
   343	  const breathe = Math.sin(time * 1.6) * s * 0.03; // ruhiges Atmen
   344	  ctx.save();
   345	  // Kontaktschatten: dicht unterm Gesaess hart, nach aussen weich
   346	  const sh = ctx.createRadialGradient(x - s * 0.2, y + s * 0.05, s * 0.1, x - s * 0.2, y + s * 0.05, s * 1.4);
   347	  sh.addColorStop(0, `rgba(0,0,0,${0.16 + 0.12 * dl})`); sh.addColorStop(1, "rgba(0,0,0,0)");
   348	  ctx.fillStyle = sh; ctx.beginPath(); ctx.ellipse(x - s * 0.2, y + s * 0.05, s * 1.4, s * 0.22, 0, 0, Math.PI * 2); ctx.fill();
   349	
   350	  // ---- Beine: je EIN durchgehender Pfad Huefte→Knie→Schienbein, der
   351	  // Stiefel waechst aus dem Bein statt daneben zu liegen. Das hintere Bein
   352	  // ist dunkler — Tiefe statt zweier identischer Teile nebeneinander.
   353	  const leg = (ox, oy, tone) => {
   354	    const hx0 = x + ox * s, hy0 = y - s * 0.5 + oy * s;      // Huefte
   355	    const kx = hx0 - s * 0.62, ky = y - s * 0.34 + oy * s;   // Knie
   356	    const ax = kx + s * 0.08, ay = y + s * 1.18 + oy * s;    // Knoechel
   357	    ctx.fillStyle = tone;

   543	// --- Draw fishing rod ---
   544	// Eine Rute ist ein KEGEL: am Griff dick, zur Spitze fein. Vorher war sie
   545	// ein Strich mit fester Breite — ausgerechnet die Menue-Vorschau
   546	// (drawRodPreview) war detaillierter als die Rute, mit der man angelt.
   547	// Jetzt gilt dieselbe Bauweise entlang der GEBOGENEN Achse: konischer
   548	// Schaft, Wicklungen, Ringe mit Steg, Korkgriff, echte Rolle mit Kurbel
   549	// (dreht beim Einholen). Die Skins wirken auf den ganzen Schaft.
   550	// ===================== Die Rute ist zurueck auf dem Blech (Runde 9) =======
   551	//
   552	// Runde 8 hat sie zu Recht heruntergenommen: im BOOT faehrt sie stufenlos
   553	// auf der Welle (469 Neubacken in 706 Bildern), und die KURBEL dreht auch im
   554	// Ruhezustand (time * 0,4 — das alte Blech backte dafuer ~12 Mal je
   555	// Sekunde neu). Beides ist jetzt ausgeschlossen: im Boot wird live
   556	// gezeichnet, und die Kurbel ist aus dem gebackenen Koerper herausgeloest —
   557	// sie sind zwei Pfade, die live ueber das Blech kommen.
   558	function drawRod() {
   559	  if (isDiveMode()) { drawHarpoon(); return; }   // Tiefsee: Harpune statt Rute
   560	  const s = dockHeight;
   561	  const bend = rodBend;
   562	  // Die Spitze wird IMMER gerechnet, auch wenn das Bild aus dem Blech kommt -
   563	  // Schnur und Schwimmer haengen daran (drawLine).
   564	  curRodTipX = lerp(rodTipX, rodTipX + (bobberX - rodTipX) * 0.25, bend);
   565	  curRodTipY = lerp(rodTipY, rodTipY + 60 * bend, bend);
   566	  const skin = getRodSkin();
   567	  const schnell = (gameState === "reeling" && isHolding) || gameState === "retrieving";
   568	  // ===================== Was sich bewegt, wird nicht gebacken ==============
   569	  // (Merksatz 12.) Im Drill biegt sich die Rute. Im BOOT faehrt sie
   570	  // stufenlos auf der Welle. `gamma` pulsiert mit der Zeit, `glow` braucht
   571	  // shadowBlur je Bild. In all diesen Faellen: live, exakt wie bisher.
   572	  if (schnell || bend > 0.002 || skin.rainbow || skin.glow || skin.fx === "gamma"
   573	      || getLocation().mode === "boat") { _drawRod(); return; }
   574	  // Ruhezustand am festen Platz: alles steht — bis auf die Kurbel. Also
   575	  // Koerper aufs Blech (rund 60 Pfade, die Haelfte Kurven), Kurbel live
   576	  // darueber. Die Signatur traegt NICHTS, was sich je Bild aendert.
   577	  const sig = [skin.id || skin.color, skin.fx || "-", skin.accent,
   578	               Math.round(rodBaseX), Math.round(rodBaseY),
   579	               Math.round(rodTipX), Math.round(rodTipY), Math.round(s)];
   580	  // Der Kasten deckt Griffende, Rolle und Spitze ab - grosszuegig, denn was
   581	  // darueber hinausragt, waere abgeschnitten.
   582	  const cx = lerp((rodBaseX + rodTipX) / 2, bobberX, bend * 0.35);
   583	  const cy = lerp((rodBaseY + rodTipY) / 2, rodTipY + 40, bend * 0.6);
   584	  const x0 = Math.min(rodBaseX, curRodTipX, cx) - s * 2.2;
   585	  const x1 = Math.max(rodBaseX, curRodTipX, cx) + s * 2.2;
   586	  const y0 = Math.min(rodBaseY, curRodTipY, cy) - s * 2.2;
   587	  const y1 = Math.max(rodBaseY, curRodTipY, cy) + s * 2.2;
   588	  Blech.bild("rute", sig, x0, y0, x1 - x0, y1 - y0, () => _drawRod(1));
   589	  _drawRod(2);   // nur die Kurbel
   590	}

   592	// `teil`: undefined = alles (der Live-Weg), 1 = nur der Koerper (fuers
   593	// Blech), 2 = nur die Kurbel (live ueber dem Blech). Die Geometrie wird in
   594	// jedem Fall gerechnet — sie ist billig, und die Kurbel sitzt an der Rolle.
   595	function _drawRod(teil) {
   596	  const skin = getRodSkin();
   597	  const s = dockHeight;
   598	  const bend = rodBend; // 0..1
   599	  const bx = rodBaseX, by = rodBaseY - s * 0.6;
   600	  const cx = lerp((rodBaseX + rodTipX) / 2, bobberX, bend * 0.35);
   601	  const cy = lerp((rodBaseY + rodTipY) / 2, rodTipY + 40, bend * 0.6);
   602	  const tipX = lerp(rodTipX, rodTipX + (bobberX - rodTipX) * 0.25, bend);
   603	  const tipY = lerp(rodTipY, rodTipY + 60 * bend, bend);
   604	
   605	  // Punkt & Normale auf der Bezier-Achse
   606	  const P = t => [(1 - t) * (1 - t) * bx + 2 * (1 - t) * t * cx + t * t * tipX,
   607	                  (1 - t) * (1 - t) * by + 2 * (1 - t) * t * cy + t * t * tipY];
   608	  const N = t => {
   609	    const dx2 = 2 * (1 - t) * (cx - bx) + 2 * t * (tipX - cx);
   610	    const dy2 = 2 * (1 - t) * (cy - by) + 2 * t * (tipY - cy);
   611	    const l = Math.hypot(dx2, dy2) || 1;
   612	    return [-dy2 / l, dx2 / l];
   613	  };
   614	  const w0 = Math.max(2.6, s * 0.16), w1 = Math.max(0.9, s * 0.05); // Kegel
   615	  // Geometrie von Griff und Rolle VOR den Zeichenbloecken: die Kurbel
   616	  // (teil 2) braucht rx/ry/rr, ohne den Koerper zu zeichnen.
   617	  const [gnx, gny] = N(0);
   618	  const gdx = bx - (P(0.14)[0] - bx), gdy = by - (P(0.14)[1] - by);   // Verlaengerung nach hinten
   619	  const rx = bx + gnx * w0 * 3.2 - (bx - gdx) * 0.35, ry = by + gny * w0 * 3.2 - (by - gdy) * 0.35;
   620	  const rr = Math.max(3, s * 0.24);
   621	
   622	  ctx.save();
   623	  if (teil !== 2) {
   624	  // Schaft als geschlossene Flaeche entlang der Kurve
   625	  if (skin.glow) { ctx.shadowColor = skin.accent; ctx.shadowBlur = 12; }
   626	  ctx.beginPath();
   627	  const SEG = 10;
   628	  for (let i = 0; i <= SEG; i++) {
   629	    const t = i / SEG, [px, py] = P(t), [nx, ny] = N(t), w = w0 + (w1 - w0) * t;
   630	    i === 0 ? ctx.moveTo(px + nx * w, py + ny * w) : ctx.lineTo(px + nx * w, py + ny * w);
   631	  }
   632	  for (let i = SEG; i >= 0; i--) {
   633	    const t = i / SEG, [px, py] = P(t), [nx, ny] = N(t), w = w0 + (w1 - w0) * t;
   634	    ctx.lineTo(px - nx * w, py - ny * w);
   635	  }
   636	  ctx.closePath();
   637	  ctx.fillStyle = rodSkinStroke(skin, bx, by, tipX, tipY);
   638	  ctx.fill();
   639	  ctx.shadowBlur = 0;
   640	  // Lichtkante auf der Oberseite des Schafts
   641	  ctx.strokeStyle = "rgba(255,255,255,0.28)"; ctx.lineWidth = Math.max(0.8, w0 * 0.35); ctx.lineCap = "round";
   642	  ctx.beginPath();
   643	  for (let i = 0; i <= SEG; i++) {
   644	    const t = i / SEG, [px, py] = P(t), [nx, ny] = N(t), w = (w0 + (w1 - w0) * t) * 0.45;
   645	    i === 0 ? ctx.moveTo(px + nx * w, py + ny * w) : ctx.lineTo(px + nx * w, py + ny * w);
   646	  }
   647	  ctx.stroke();
   648	  drawRodPattern(skin, bx, by, cx, cy, tipX, tipY);
   649	
   650	  // Ringe: sitzen auf einem kurzen Steg UNTER dem Schaft, folgen der
   651	  // Kruemmung, werden zur Spitze kleiner - dort laeuft die Schnur durch
   652	  ctx.strokeStyle = skin.accent; ctx.lineCap = "round";
   653	  for (const t of [0.3, 0.5, 0.68, 0.84, 0.97]) {
   654	    const [px, py] = P(t), [nx, ny] = N(t);
   655	    const r = Math.max(1.2, s * 0.11 * (1 - t * 0.6));
   656	    const st = Math.max(1.5, s * 0.08);
   657	    ctx.lineWidth = Math.max(1, s * 0.045);
   658	    ctx.beginPath(); ctx.moveTo(px - nx * (w0 * 0.2), py - ny * (w0 * 0.2)); ctx.lineTo(px - nx * (st + r), py - ny * (st + r)); ctx.stroke();
   659	    ctx.lineWidth = Math.max(1, s * 0.055);
   660	    ctx.beginPath(); ctx.arc(px - nx * (st + r), py - ny * (st + r), r, 0, Math.PI * 2); ctx.stroke();
   661	    // Wicklung am Stegfuss
   662	    ctx.lineWidth = Math.max(1, s * 0.05);
   663	    ctx.beginPath(); ctx.moveTo(px + nx * (w0 + (w1 - w0) * t), py + ny * (w0 + (w1 - w0) * t)); ctx.lineTo(px - nx * (w0 + (w1 - w0) * t), py - ny * (w0 + (w1 - w0) * t)); ctx.stroke();
   664	  }
   665	
   666	  // Korkgriff mit Endkappe, entlang der Achse unter dem Ansatz
   667	  ctx.strokeStyle = "#b98a5a"; ctx.lineWidth = w0 * 2.1; ctx.lineCap = "round";
   668	  ctx.beginPath(); ctx.moveTo(bx, by); ctx.lineTo(gdx, gdy); ctx.stroke();
   669	  ctx.strokeStyle = "rgba(60,35,18,0.4)"; ctx.lineWidth = w0 * 2.1;
   670	  ctx.setLineDash([Math.max(1, s * 0.05), Math.max(2, s * 0.14)]);
   671	  ctx.beginPath(); ctx.moveTo(bx, by); ctx.lineTo(gdx, gdy); ctx.stroke();
   672	  ctx.setLineDash([]);
   673	  ctx.strokeStyle = "#2b2b30"; ctx.lineWidth = w0 * 2.4; // Endkappe
   674	  ctx.beginPath(); ctx.moveTo(gdx, gdy); ctx.lineTo(gdx - (bx - gdx) * 0.12, gdy - (by - gdy) * 0.12); ctx.stroke();
   675	
   676	  // Rolle: Gehaeuse, Spule in Akzentfarbe, Kurbel dreht beim Einholen
   677	  ctx.strokeStyle = "#3a3f47"; ctx.lineWidth = Math.max(1.5, s * 0.08);
   678	  ctx.beginPath(); ctx.moveTo(bx + gnx * w0, by + gny * w0); ctx.lineTo(rx, ry); ctx.stroke();  // Fuss
   679	  const rg = ctx.createRadialGradient(rx - rr * 0.3, ry - rr * 0.3, rr * 0.2, rx, ry, rr);
   680	  rg.addColorStop(0, "#5a616b"); rg.addColorStop(1, "#23262c");
   681	  ctx.fillStyle = rg; ctx.beginPath(); ctx.arc(rx, ry, rr, 0, Math.PI * 2); ctx.fill();
   682	  ctx.fillStyle = skin.accent; ctx.beginPath(); ctx.arc(rx, ry, rr * 0.45, 0, Math.PI * 2); ctx.fill();
   683	  } // Ende Koerper (teil !== 2)
   684	  if (teil !== 1) {
   685	  // Die Kurbel: das EINZIGE an der ruhenden Rute, das sich je Bild bewegt.
   686	  // Deshalb kommt sie live ueber das gebackene Bild (drawRod, Runde 9).
   687	  const spin = (gameState === "reeling" && isHolding) || gameState === "retrieving" ? time * 14 : time * 0.4;
   688	  ctx.strokeStyle = "#c9ced6"; ctx.lineWidth = Math.max(1.2, s * 0.06); ctx.lineCap = "round";
   689	  ctx.beginPath(); ctx.moveTo(rx, ry); ctx.lineTo(rx + Math.cos(spin) * rr * 1.15, ry + Math.sin(spin) * rr * 1.15); ctx.stroke();
   690	  ctx.fillStyle = "#e8e2d2";
   691	  ctx.beginPath(); ctx.arc(rx + Math.cos(spin) * rr * 1.15, ry + Math.sin(spin) * rr * 1.15, Math.max(1.2, s * 0.07), 0, Math.PI * 2); ctx.fill();
   692	  }
   693	  ctx.restore();
   694	}

   935	function drawHook() {
   936	  ctx.save();
   937	  ctx.strokeStyle = "rgba(200,220,240,0.6)";
   938	  ctx.lineWidth = 1;
   939	  // Unter Wasser: zum Haken – im Drill zum Maul des Fisches, mit Bogen in Zugrichtung
   940	  const tx = hookedFish && gameState === "reeling" ? hookedFish.x + hookedFish.dir * fishUnit(hookedFish) * 0.9 : hookX;
   941	  const ty = hookedFish && gameState === "reeling" ? hookedFish.y : hookY;
   942	  ctx.beginPath(); ctx.moveTo(bobberX, bobberY + 6);
   943	  if (gameState === "reeling") ctx.quadraticCurveTo(bobberX + lineBow * 0.6, (bobberY + ty) / 2, tx, ty); else ctx.lineTo(tx, ty);
   944	  ctx.stroke();
   945	  if (gameState === "reeling") { ctx.restore(); return; } // Haken steckt im Maul – nicht extra zeichnen
   946	  ctx.strokeStyle = "#cfd6dd"; ctx.lineWidth = 2; ctx.lineCap = "round";
   947	  const s = 5 * uiScale();
   948	  ctx.beginPath(); ctx.moveTo(hookX, hookY); ctx.lineTo(hookX, hookY + s * 1.6);
   949	  ctx.arc(hookX - s * 0.8, hookY + s * 1.6, s * 0.8, 0, Math.PI, false); ctx.stroke();
   950	  // Koeder am Haken.
   951	  //
   952	  // ===================== Die Emoji-Schrift steht VORNE (01.09.) ============
   953	  //
   954	  // Dustin: "fehlende Koeder, die werden einfach nicht mehr dargestellt."
   955	  // Hier ist er nicht nachzustellen - an allen sechs Orten, mit Haken im
   956	  // Wasser, in Stufe VOLL wie in SPARSAM, zeichnet der Wurm. Was sich aber
   957	  // benennen laesst, ist die einzige Stelle, an der dieses Bild von einem
   958	  // FALLBACK abhaengt statt von einer Angabe:
   959	  //
   960	  // `font.js` haengt sich in den font-Setter und ersetzt `sans-serif` durch
   961	  // die Hausschrift (Baloo 2, aus einer Data-URI geladen). Der Kommentar dort
   962	  // sagt: "Emoji haben in Baloo 2 keine Glyphe - der Browser faellt dort von
   963	  // selbst auf die Emoji-Schrift zurueck." Auf einem Entwicklungsrechner
   964	  // stimmt das. Ob WebKit aus einer per FontFace nachgeladenen Schrift heraus
   965	  // im CANVAS zuverlaessig auf Apple Color Emoji zurueckfaellt, ist genau die
   966	  // Sorte Annahme, die dieses Projekt schon dreimal Zeit gekostet hat.
   967	  //
   968	  // Also wird nichts mehr angenommen: die Emoji-Schrift steht jetzt VORN.
   969	  // `font.js` haengt seine Familien hinter dem `sans-serif` am Ende weiterhin
   970	  // an, der Weg zur Hausschrift bleibt also offen - er ist nur nicht mehr der
   971	  // erste. Kostet nichts und nimmt eine Unbekannte heraus.
   972	  const bait = getBait();
   973	  ctx.font = `${Math.round(10 * uiScale())}px "Apple Color Emoji","Segoe UI Emoji","Noto Color Emoji",sans-serif`;
   974	  ctx.textAlign = "center"; ctx.textBaseline = "middle";
   975	  ctx.fillText(bait.icon, hookX - s * 1.6, hookY + s * 1.8);
   976	  ctx.restore();
   977	}

  1496	// Damit ging der Effekt verloren, sobald sich ein Element zwischen Druck und Zeichnen
  1497	// bewegt – und das tut ab jetzt jedes Element, weil das Menü von unten hochfährt.
  1498	// Jetzt trägt jede Trefffläche einen SCHLÜSSEL, und der Effekt hängt am Schlüssel.

  1502	// Wie sich ein Brett unter dem Daumen verhält: es gibt kurz nach und federt EINMAL
  1503	// gedämpft zurück. Kein Gummi (Overshoot ~14 % der Eindrücktiefe), kein Nachwippen.
  1504	const PRESS_DUR = 0.24;
  1505	function pressWave(t) {
  1506	  if (t < 0.06) return t / 0.06;                       // 60 ms hinein
  1507	  const u = (t - 0.06) / (PRESS_DUR - 0.06);           // 180 ms zurück, gedämpft
  1508	  if (u >= 1) return 0;
  1509	  return Math.exp(-3.4 * u) * Math.cos(4.9 * u);
  1510	}

  1755	function fitText(text, maxWidth) {
  1756	  // Erst übersetzen, dann kürzen. Vorher wurde die DEUTSCHE Zeichenkette gekürzt,
  1757	  // während gegen die ENGLISCHE Breite gemessen wurde (measureText übersetzt mit).
  1758	  // Ergebnis: im englischen Spiel stand "Holzrut…" statt "Wooden ro…".
  1759	  text = I18N.translate(text);
  1760	  if (ctx.measureText(text).width <= maxWidth) return text;
  1761	  let t = text;
  1762	  while (t.length > 1 && ctx.measureText(t + "…").width > maxWidth) t = t.slice(0, -1);
  1763	  return t + "…";
  1764	}
  1765	
  1766	// Bricht an Wortgrenzen auf mehrere Zeilen um; die letzte Zeile wird bei Bedarf gekürzt.
  1767	// Auf dem Handy passt eine Beschreibung sonst nicht in eine Zeile und endet als "…".
  1768	function wrapText(text, maxWidth, maxLines = 2) {
  1769	  // Erst übersetzen, dann umbrechen. Sonst zerlegt der Umbruch den Satz und die
  1770	  // Wörterbuchzeile für den ganzen Satz greift nicht mehr – deshalb standen im
  1771	  // englischen Shop und in den Talenten deutsche Beschreibungen.
  1772	  const words = String(I18N.translate(text)).split(" ");
  1773	  const lines = [];
  1774	  let cur = "";
  1775	  for (const wd of words) {
  1776	    const test = cur ? cur + " " + wd : wd;
  1777	    if (ctx.measureText(test).width <= maxWidth || !cur) cur = test;
  1778	    else { lines.push(cur); cur = wd; if (lines.length === maxLines - 1) break; }
  1779	  }
  1780	  if (lines.length < maxLines) {
  1781	    const rest = words.slice(lines.join(" ").split(" ").filter(Boolean).length).join(" ");
  1782	    lines.push(fitText(lines.length === maxLines - 1 ? (rest || cur) : cur, maxWidth));
  1783	  }
  1784	  return lines.slice(0, maxLines);
  1785	}

  1871	    const need = top + count * rowH + bot;
  1872	    return count > 0 && need <= avail ? need : F;
  1873	  };
  1874	  switch (id) {
  1875	    case "daily": {
  1876	      const cellH = 92 * s, btnH = 58 * s, noteH = 46 * s, head = narrow ? 0 : 14 * s;
  1877	      return head + 64 * s + 22 * s + cellH + 26 * s + btnH + 14 * s + btnH + 26 * s + noteH + 16 * s;

  1982	  const key = overlay ? ((overlayGroupOf(overlay) || {}).id || overlay) : null;
  1983	  if (key !== overlayAnimKey) { overlayAnimKey = key; overlayAnim = 0; }   // Reiterwechsel schiebt NICHT neu hoch
  1984	  if (!overlay) return 0;
  1985	  // Höhe weich nachführen: beim Öffnen sofort, beim Reiterwechsel über ~8 Bilder.
  1986	  const t = overlayTargetH();
  1987	  if (!overlaySheetH || overlayAnim === 0) overlaySheetH = t;
  1988	  else overlaySheetH += (t - overlaySheetH) * 0.22;
  1989	  if (Math.abs(t - overlaySheetH) < 0.6) overlaySheetH = t;
  1990	  overlayAnim = Math.min(1, overlayAnim + 0.08);   // ~12 Bilder ≈ 200 ms
  1991	  return overlayAnim;

  1996	function overlayRiseOffset(t, h) {
  1997	  if (t >= 1) return 0;
  1998	  const k = 1 - Math.pow(1 - t, 3);
  1999	  const bump = t < 0.55 ? 0 : Math.sin((t - 0.55) / 0.45 * Math.PI) * 5;
  2000	  return (h + 24) * (1 - k) - bump;
  2001	}

  2354	function drawRodPreview(item, x, y, s) {
  2355	  const up = rodUpgradeLevel(item.id);
  2356	  const bx = x - 13 * s, by = y + 14 * s;      // Griffende
  2357	  const tx = x + 15 * s, ty = y - 18 * s;      // Spitze
  2358	  const dx = tx - bx, dy = ty - by, len = Math.hypot(dx, dy);
  2359	  const nx = -dy / len, ny = dx / len;         // Normale zur Rutenachse
  2360	  const at = (t, off) => [bx + dx * t + nx * off, by + dy * t + ny * off];
  2361	
  2362	  if (item.glow) { ctx.shadowColor = item.glow; ctx.shadowBlur = 8 * s; }
  2363	  // Schaft als Trapez (dick am Griff, dünn zur Spitze)
  2364	  const w0 = 2.6 * s, w1 = 0.5 * s;
  2365	  ctx.beginPath();
  2366	  ctx.moveTo(...at(0.16, w0)); ctx.lineTo(...at(1, w1)); ctx.lineTo(...at(1, -w1)); ctx.lineTo(...at(0.16, -w0));
  2367	  ctx.closePath();
  2368	  const g = ctx.createLinearGradient(bx, by, tx, ty);
  2369	  g.addColorStop(0, item.shaft || "#8a8f96");
  2370	  g.addColorStop(0.5, shadeColor(item.shaft || "#8a8f96", 22));
  2371	  g.addColorStop(1, shadeColor(item.shaft || "#8a8f96", -10));
  2372	  ctx.fillStyle = g; ctx.fill();
  2373	  ctx.shadowBlur = 0;
  2374	
  2375	  // Teleskop-Segmente
  2376	  if (item.segments) {
  2377	    ctx.strokeStyle = "rgba(0,0,0,0.35)"; ctx.lineWidth = 1 * s;
  2378	    for (const t of [0.42, 0.62, 0.8]) {
  2379	      const wt = w0 + (w1 - w0) * ((t - 0.16) / 0.84);
  2380	      ctx.beginPath(); ctx.moveTo(...at(t, wt)); ctx.lineTo(...at(t, -wt)); ctx.stroke();
  2381	    }
  2382	  }
  2383	  // Ringe
  2384	  ctx.strokeStyle = item.accent || "#dfe6ee"; ctx.lineWidth = 1.2 * s;
  2385	  for (const t of [0.5, 0.72, 0.92]) {
  2386	    const [rx, ry] = at(t, 0);
  2387	    ctx.beginPath(); ctx.arc(rx, ry, 2.2 * s, 0, Math.PI * 2); ctx.stroke();
  2388	  }
  2389	  // Griff
  2390	  ctx.strokeStyle = item.grip || "#4a4f57"; ctx.lineWidth = 5.5 * s; ctx.lineCap = "round";
  2391	  ctx.beginPath(); ctx.moveTo(...at(-0.02, 0)); ctx.lineTo(...at(0.2, 0)); ctx.stroke();
  2392	  // Rolle
  2393	  const [cx, cy] = at(0.26, -4.5 * s);
  2394	  ctx.fillStyle = shadeColor(item.grip || "#4a4f57", 30);
  2395	  ctx.beginPath(); ctx.arc(cx, cy, 4.2 * s, 0, Math.PI * 2); ctx.fill();
  2396	  ctx.fillStyle = item.accent || "#dfe6ee";
  2397	  ctx.beginPath(); ctx.arc(cx, cy, 1.7 * s, 0, Math.PI * 2); ctx.fill();
  2398	
  2399	  // Reichweitenbalken – Grundwert weiß, Upgrade-Anteil in Akzentfarbe
  2400	  const barW = 30 * s, barX = x - barW / 2, barY = y + 22 * s;
  2401	  const maxR = RODS[RODS.length - 1].radius;
  2402	  const base = clamp(item.radius / maxR, 0, 1);
  2403	  const eff = clamp(item.radius * (1 + 0.03 * up) / maxR, 0, 1);

  2405	  if (up) {

  2410	  ctx.beginPath(); ctx.roundRect(barX, barY, barW * base, 3.5 * s, 2 * s); ctx.fill();
  2411	}

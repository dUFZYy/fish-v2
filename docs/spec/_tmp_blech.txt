     6	// RUNDE 9 (01.09.2026): DAS BLECH KOMMT ZURUECK — MIT ZWEI SICHERUNGEN
     7	// -----------------------------------------------------------------------------
     8	//
     9	// Runde 8 hat das Blech vom Angelplatz genommen, weil es dort NEU BACKTE statt
    10	// zu treffen (Boot 3,38 / Riff 3,50 / Tiefsee 6,00 Neubacken je Bild). Die
    11	// Diagnose war richtig, die Therapie zu grob: der Fehler waren instabile
    12	// SIGNATUREN, nicht das Caching. Die Geraetezahlen sagen das deutlich —
    13	// mit Blech lief der See bei 66,7 fps, ohne bei 43,5 (gleiche Aufloesung;
    14	// Akkustand verschieden, deshalb entscheidet der Duell-Lauf, bench.js).
    15	//
    16	// Zurueck kommt NUR, was eine stabile Signatur hat:
    17	//
    18	//   * die sechs KULISSEN (zweiteilig: Grundriss aufs Blech, was weht live).
    19	//     Ihre Signatur haengt an Ort, Lichtstufe (1/64) und Bildgroesse —
    20	//     ein Neubacken alle paar Sekunden in der Daemmerung, sonst nie.
    21	//   * die RUTE im Ruhezustand — aber OHNE die Kurbel. Die dreht mit
    22	//     time*0,4 und hat das alte Blech mit ~12 Neubacken je Sekunde
    23	//     totgedreht. Jetzt: Koerper gebacken, Kurbel live (zwei Pfade).
    24	//   * das PLANKTON der Tiefsee als Stempel — diesmal rastern Schluessel,
    25	//     Signatur UND ZEICHNUNG auf DENSELBEN Wert (Merksatz 13). Die alte
    26	//     Fassung fasste im Schluessel zusammen, was die Signatur trennte.
    27	//   * das LEUCHTEN der Tiefsee-Korallen und des Monds: der teuerste Posten
    28	//     dort war nie die Flaeche, sondern `shadowBlur` — ein Gauss-Weichzeichner
    29	//     JE ZEICHENBEFEHL, jedes Bild, ~44 Mal in der Tiefsee. Der Halo wird
    30	//     jetzt einmal gebacken (mit exakt demselben shadowBlur) und danach
    31	//     kopiert; die scharfen Konturen bleiben live und bewegen sich weiter.
    32	//
    33	// NICHT zurueck kommen: Boot (faehrt stufenlos auf der Welle), Angler (atmet
    34	// stufenlos), Wolken, Voegel, Quallen. Was sich stufenlos bewegt, gehoert
    35	// nicht auf ein Blech, das Unterpixel mitbackt (Merksatz 12).
    36	//
    37	// DIE ZWEI SICHERUNGEN, die Runde 8 unmoeglich machen:
    38	//
    39	//   1. NEUBACK-BUDGET JE BILD. Ein Bild darf hoechstens EIN grosses Motiv
    40	//      und ein paar kleine Stempel neu backen. Wer sein Budget nicht
    41	//      bekommt, zeichnet in diesem Bild DIREKT (der alte Preis, kein
    42	//      Aufschlag) und backt im naechsten. Damit degradiert jeder kuenftige
    43	//      Signatur-Fehler zu "kein Cache" statt zu "Anti-Cache mit
    44	//      126-ms-Bildern" — genau die Kostenart, die Runde 8 gefunden hat.
    45	//   2. NEUBACKEN JE BILD WIRD IMMER GEZAEHLT. `Blech.jeBild()` liefert die
    46	//      Rate seit dem letzten Ablesen; der Pruefstand druckt sie je Zeile.
    47	//      Ein Motiv, das oefter als in einem von hundert Bildern backt,
    48	//      fliegt wieder runter (Merksatz 14).
    49	//
    50	// -----------------------------------------------------------------------------
    51	// WOFUER
    52	// -----------------------------------------------------------------------------
    53	//
    54	// Dieses Spiel zeichnet alles prozedural und jedes Bild neu. Am See sind das
    55	// rund 1400 Zeichenbefehle allein fuer die Kulisse — Berge, Huegelzug,
    56	// Nadelwald, Uferkante —, und keiner davon malt in zwei aufeinanderfolgenden
    57	// Bildern etwas anderes. Dieselbe Rechnung gilt fuer die Bretterwand im
    58	// Becken (760 Befehle), das Regal (312) und den Unterschrank (423).
    59	//
    60	// Genau dafuer ist ein Sprite-System da: **einmal rastern, dann kopieren.**
    61	// Der ganze Zeichencode bleibt, wie er ist; er laeuft nur nicht mehr in jedes
    62	// Bild, sondern in eines von hundert.
    63	//
    64	// Das geht hier ueberhaupt nur, weil jede Zeichenfunktion im Projekt
    65	// DETERMINISTISCH ist: `prnd(i, salt)` streut absichtlich nicht zur Laufzeit,
    66	// sondern liefert fuer denselben Index immer denselben Wert.
    67	//
    68	// -----------------------------------------------------------------------------
    69	// DIE REGELN, JEDE MIT EINER MESSUNG DAHINTER (docs/LEISTUNG-BASIS.md)
    70	// -----------------------------------------------------------------------------
    71	//
    72	// 1. NIE SKALIERT BLITTEN. Gebacken wird in Geraetepixeln, in genau der
    73	//    Groesse, in der es steht; die Unterpixel-Lage wird mitgebacken und
    74	//    steht in der Signatur (sonst sitzt jede Kante einen halben Pixel
    75	//    daneben — 6823 abweichende Pixel im Diff, mit Lage 533).
    76	// 2. NIE IM SELBEN BILD SCHREIBEN UND LESEN — genau einmal je Neubacken,
    77	//    und Neubacken sind budgetiert (oben).
    78	// 3. KEIN UMWEG JE BENUTZUNG. Der Treffer ist ein drawImage, sonst nichts.
    79	// 4. GEGEN "AUS" MESSEN, AUF DEM GERAET. `Blech.an = false` schaltet alles
    80	//    still; der Duell-Lauf (bench.js) misst an/aus VERSCHRAENKT im selben
    81	//    Lauf — nur solche Vergleiche zaehlen.
    82	// 5. NEUBACKEN JE BILD ZAEHLEN, nicht Trefferquoten je Sitzung.
    83	//
    84	// WAS NICHT AUF DAS BLECH DARF: Text (i18n.js uebersetzt beim Zeichnen),
    85	// alles, was sich jedes Bild aendert, Druck-Rueckmeldung/Trefferflaechen
    86	// (die Aufrufer backen die Kulisse eines Bedienelements, nie das Element).
    87	//
    88	// SPEICHER: Deckel `MAX_BYTES`, LRU (Einfuegereihenfolge der Map).
    89	// `Blech.bericht()` sagt, was wirklich drinliegt.
    90	// =============================================================================
    91	
    92	const Blech = {
    93	  // Aus heisst: `bild()` zeichnet direkt und legt nichts an. Der Schalter ist
    94	  // fuer die Gegenprobe im Pruefstand da (Merksatz 8) und fuer den Notfall.
    95	  an: true,
    96	
    97	  // Der Deckel. Er muss die beiden Becken-Motive auf JEDEM Schirm halten:
    98	  // Bretterwand und Werkbank sind gemessen 9,2 + 3,3 MB bei 390x844 @3x und
    99	  // rund 15 MB auf einem 430x932-Schirm; dazu kommen die Kulisse des
   100	  // aktiven Ortes (2-5 MB) und eine Handvoll Stempel. Ein Deckel, der
   101	  // Motive abwechselnd hinauswirft, wuerde sie in jedem Bild neu backen.
   102	  MAX_BYTES: 32 * 1024 * 1024,
   103	
   104	  // Ueber dieser Kantenlaenge wird gar nicht erst gebacken. Schuetzt vor dem
   105	  // Fall, dass ein Aufrufer versehentlich etwas Riesiges anmeldet.
   106	  MAX_KANTE: 4096,
   107	  MAX_PIXEL: 6 * 1024 * 1024,   // ~24 MB in EINEM Motiv waeren zu viel
   108	  // Geraetepixel Kante fuer Stempel. 512 statt 192 (Runde 7): die Halo-
   109	  // Stempel der Tiefsee-Korallen brauchen bis ~370 px Kante. Ein 512er
   110	  // Stempel ist 1 MB — gespart wird dort kein Fuellwerk (Merksatz 10),
   111	  // sondern der Gauss-Weichzeichner je Bild.
   112	  MAX_STEMPEL: 512,
   113	
   114	  // ===================== Das Neuback-Budget (Runde 9) ======================
   115	  //
   116	  // Ein Neubacken ist eine Leinwand, die im selben Bild beschrieben und
   117	  // wieder gelesen wird — die teuerste Kostenart dieses Projekts (11 von
   118	  // 16,7 ms, LEISTUNG-BASIS Regel 1). Das Budget deckelt sie JE BILD:
   119	  //
   120	  //   * hoechstens EIN grosses Motiv (`bild`) — ein Kulissen-Neubacken
   121	  //     kostet genau den alten Je-Bild-Preis, zwei davon kosteten einen
   122	  //     Ruckler.
   123	  //   * hoechstens VIER kleine Stempel — ihre Leinwaende sind winzig
   124	  //     (< 256 px Kante), der erzwungene Flush betrifft nur ihre eigenen
   125	  //     paar Befehle.
   126	  //
   127	  // Wer nicht drankommt, zeichnet DIREKT (alter Preis) und ist im naechsten
   128	  // Bild wieder Kandidat. Beim Betreten eines Ortes backt sich das Blech so
   129	  // ueber die ersten Bilder gestaffelt voll, statt den Ortswechsel-Frame zu
   130	  // sprengen. Und ein Signatur-Fehler wie in Runde 8 (Boot: jede
   131	  // Unterpixel-Lage eine neue Signatur) kostet ab jetzt hoechstens den
   132	  // Preis von "Blech aus" — nie mehr 126-ms-Bilder.
   133	  BUDGET_BILD: 1,
   134	  BUDGET_STEMPEL: 4,
   135	  _budgetBild: 1,
   136	  _budgetStempel: 4,
   137	  _frameStamp: -1,
   138	
   139	  _m: new Map(),      // key -> Eintrag; Einfuegereihenfolge ist die LRU-Ordnung
   140	  bytes: 0,
   141	  stat: { treffer: 0, backen: 0, direkt: 0, raus: 0, aufschub: 0 },
   142	
   143	  // Zaehlt Bilder fuer `jeBild()` — der Stempel kommt von `lastFrame`
   144	  // (script.js), das sich genau einmal je Bild aendert.
   145	  _bilder: 0,
   146	  _lesBacken: 0, _lesBilder: 0,   // Stand beim letzten Ablesen
   147	
   148	  _neuesBild() {
   149	    const f = (typeof lastFrame !== "undefined") ? lastFrame : -1;
   150	    if (f === this._frameStamp) return;
   151	    this._frameStamp = f;
   152	    this._bilder++;
   153	    this._budgetBild = this.BUDGET_BILD;
   154	    this._budgetStempel = this.BUDGET_STEMPEL;
   155	  },
   156	
   157	  // Neubacken je Bild seit dem letzten Ablesen. DIE Zahl, an der ein Cache
   158	  // haengt (Merksatz 14): ueber 0,01 ist ein Motiv faul, ab 1 ist es ein
   159	  // Anti-Cache. Der Duell-Lauf druckt sie je Zeile.
   160	  jeBild(reset) {
   161	    const dB = this.stat.backen - this._lesBacken;
   162	    const dF = this._bilder - this._lesBilder;
   163	    if (reset) { this._lesBacken = this.stat.backen; this._lesBilder = this._bilder; }
   164	    return dF > 0 ? dB / dF : 0;
   165	  },
   166	
   167	  // --- Eine gerasterte Zahl fuer die Signatur -------------------------------
   168	  //
   169	  // Fuer Werte, die sich LANGSAM aendern und trotzdem in eine Signatur
   170	  // muessen (Tageslicht in 64 Stufen). `schritt` so waehlen, dass ein
   171	  // Schritt unter der Wahrnehmungsschwelle liegt. NIE fuer etwas benutzen,
   172	  // das sich jedes Bild aendert — das ist der Fehler von Runde 8.
   173	  raster(v, schritt) { return Math.round(v / schritt); },
   174	
   175	  // --- Die zweigeteilten Kulissen -------------------------------------------
   176	  //
   177	  // Die Kulissen bestehen aus einem grossen unbewegten Grundriss und ein paar
   178	  // Teilen, die wehen (See: Berge, Huegelzug, Nadelwald, Uferkante — plus 60
   179	  // Grashalme und 12 Schilfrohre). Dieselbe Zeichenfunktion laeuft zweimal:
   180	  // im ersten Lauf ist `teil = RUHIG`, sie ueberspringt ihre bewegten
   181	  // Bloecke und landet auf dem Blech; im zweiten ist `teil = BEWEGT`, sie
   182	  // zeichnet nur noch live, was weht.
   183	  //
   184	  // BEDINGUNG, und sie ist ernst: die bewegten Teile muessen in der
   185	  // Zeichenreihenfolge OBEN liegen. Was hinter dem Grundriss steht (das
   186	  // Polarlicht der Arktis), gehoert nicht in dieses Verfahren — es kaeme
   187	  // sonst nach vorn. Die Arktis ist deshalb von Hand geteilt.
   188	  ALLES: 0, RUHIG: 1, BEWEGT: 2,
   189	  teil: 0,
   190	
   191	  // In einer Zeichenfunktion: `if (Blech.ruhig()) { ...Grundriss... }`
   192	  ruhig() { return this.teil !== 2; },
   193	  // und `if (Blech.bewegt()) { ...was weht... }`
   194	  bewegt() { return this.teil !== 1; },
   195	
   196	  zweiteilig(key, sig, x, y, w, h, mal) {
   197	    if (!this.an) { this.teil = 0; this.stat.direkt++; mal(); return; }
   198	    try {
   199	      this.teil = this.RUHIG;
   200	      this.bild(key, sig, x, y, w, h, mal);
   201	      this.teil = this.BEWEGT;
   202	      mal();
   203	    } finally { this.teil = this.ALLES; }
   204	  },
   205	
   206	  // --- Der Aufruf ----------------------------------------------------------
   207	  //
   208	  //   Blech.bild(key, sig, x, y, w, h, mal)
   209	  //
   210	  //   key   Name des Motivs. Ein Motiv je Name; wechselt die Signatur, wird
   211	  //         dasselbe Motiv neu gebacken (kein zweiter Eintrag).
   212	  //   sig   Alles, wovon das AUSSEHEN abhaengt. Array oder String.
   213	  //         Was hier fehlt, friert im Bild ein. Was hier zu viel steht,
   214	  //         macht den Cache wertlos.
   215	  //   x,y,w,h  Der Bereich, den das Motiv belegt — in den Koordinaten, in
   216	  //         denen `mal` zeichnet (also CSS-Pixel, wie der ganze Zeichencode).
   217	  //         Grosszuegig angeben: was darueber hinausragt, wird abgeschnitten.
   218	  //   mal() Zeichnet das Motiv. Nimmt keine Argumente und benutzt den
   219	  //         globalen `ctx` wie jede andere Zeichenfunktion im Projekt —
   220	  //         waehrend des Backens zeigt der auf die Blech-Leinwand.
   221	  //
   222	  // Rueckgabe: true, wenn aus dem Cache kopiert wurde.
   223	  bild(key, sig, x, y, w, h, mal) {
   224	    this._neuesBild();
   225	    if (!this.an) { this.stat.direkt++; mal(); return false; }
   226	    const c = ctx;
   227	    let t;
   228	    try { t = c.getTransform(); } catch (e) { this.stat.direkt++; mal(); return false; }
   229	    // Gedreht oder geschert laesst sich nicht pixelgenau kopieren. Kommt im
   230	    // Projekt an diesen Stellen nicht vor; wenn doch, wird direkt gezeichnet.
   231	    if (Math.abs(t.b) > 1e-6 || Math.abs(t.c) > 1e-6 || t.a <= 0 || t.d <= 0) {
   232	      this.stat.direkt++; mal(); return false;
   233	    }
   234	    const sx = t.a, sy = t.d;
   235	    // Geraetepixel: das Motiv beginnt bei dx0/dy0 und ist pw x ph gross.
   236	    // Aufgerundet, damit die rechte und untere Kante vollstaendig drin sind.
   237	    const gx = x * sx + t.e, gy = y * sy + t.f;
   238	    const dx0 = Math.round(gx), dy0 = Math.round(gy);
   239	    // ============ Die halbe Pixel, die man sieht ============
   240	    //
   241	    // Kopiert wird auf ganze Geraetepixel (alles andere waere eine
   242	    // Neuabtastung und damit weich). Gezeichnet wird aber an der exakten,
   243	    // meist gebrochenen Stelle. Also wird der Bruchteil MITGEBACKEN, und er
   244	    // gehoert in die Signatur (auf ein Viertelpixel gerastert). DARAUS folgt
   245	    // Merksatz 12: was sich stufenlos bewegt, darf hier nicht herein — seine
   246	    // Lage waere jedes Bild eine neue Signatur.
   247	    const fx = gx - dx0, fy = gy - dy0;
   248	    const pw = Math.ceil(w * sx) + 2, ph = Math.ceil(h * sy) + 2;
   249	    if (pw < 2 || ph < 2 || pw > this.MAX_KANTE || ph > this.MAX_KANTE || pw * ph > this.MAX_PIXEL) {
   250	      this.stat.direkt++; mal(); return false;
   251	    }
   252	
   253	    // Die Groesse in Geraetepixeln gehoert IMMER in die Signatur — sonst
   254	    // ueberlebt ein Motiv einen Wechsel der Aufloesung (Drehen des Geraets,
   255	    // Stufenwechsel) und wird danach skaliert kopiert. Das ist Regel 1.
   256	    const s = (Array.isArray(sig) ? sig.join("") : String(sig))
   257	            + "" + pw + "x" + ph
   258	            + "" + Math.round(fx * 4) + "," + Math.round(fy * 4);
   259	
   260	    let e = this._m.get(key);
   261	    if (e && e.s === s) {
   262	      this._m.delete(key); this._m.set(key, e);   // ans Ende: zuletzt benutzt
   263	      this.stat.treffer++;
   264	    } else {
   265	      // Neubacken — aber nur, wenn das Budget dieses Bildes es hergibt.
   266	      // Sonst: direkt zeichnen (der alte Preis) und im naechsten Bild wieder
   267	      // anstehen. Siehe "Das Neuback-Budget" oben.
   268	      if (this._budgetBild <= 0) { this.stat.aufschub++; this.stat.direkt++; mal(); return false; }
   269	      this._budgetBild--;
   270	      e = this._backe(key, e, s, pw, ph, x, y, sx, sy, fx, fy, mal);
   271	      if (!e) { this.stat.direkt++; mal(); return false; }
   272	    }
   273	
   274	    // Kopieren: Transformation zurueck auf Geraetepixel, 1:1, keine
   275	    // Glaettung noetig, weil Quelle und Ziel dieselbe Aufloesung haben.
   276	    c.save();
   277	    c.setTransform(1, 0, 0, 1, 0, 0);
   278	    // Alpha und Composite des Aufrufers bleiben absichtlich stehen: ein
   279	    // Motiv darf halbdurchsichtig oder additiv kopiert werden.
   280	    c.drawImage(e.cv, dx0, dy0);
   281	    c.restore();
   282	    return true;
   283	  },
   284	
   285	  // --- Der Stempel: ein kleines Motiv, das oft und ueberall steht ----------
   286	  //
   287	  //   Blech.stempel(key, sig, x, y, hw, hh, mal)
   288	  //
   289	  // Fuer Leuchtpunkte, Halos, weiche Glows: 55 Planktonpunkte der Tiefsee
   290	  // sind sonst 55 `createRadialGradient` JE BILD; das Korallen-Leuchten
   291	  // sind sonst ~44 `shadowBlur`-Zeichenbefehle JE BILD (je einer ein
   292	  // Gauss-Durchlauf im Browser).
   293	  //
   294	  // Unterschied zu `bild()`: der Stempel wird NICHT auf ganze Geraetepixel
   295	  // gerastert. Er steht an der exakten, gebrochenen Stelle, und der Browser
   296	  // rechnet die Unterpixel-Lage aus. Bei einem weichen Verlauf ist das genau
   297	  // richtig — er hat keine Kante, an der man eine Neuabtastung saehe. Fuer
   298	  // harte Kanten waere `bild()` die richtige Wahl.
   299	  //
   300	  // GROESSE BLEIBT GROESSE: kopiert wird 1:1, das Motiv wird nie gestreckt.
   301	  //
   302	  // HELLIGKEIT ueber `globalAlpha` des Aufrufers. Das ist erlaubt und exakt,
   303	  // solange ALLE Alphawerte des Motivs proportional zur gewuenschten
   304	  // Helligkeit sind — dann ist "einmal bei voller Helligkeit backen und mit
   305	  // Alpha kopieren" dasselbe Bild. Wo das nicht gilt, gehoert die
   306	  // Helligkeit in die Signatur.
   307	  //
   308	  // MERKSATZ 13 GILT HIER WOERTLICH: was im Schluessel zusammengefasst wird,
   309	  // muss auch GEZEICHNET werden. Der Aufrufer rastert seinen Wert (etwa den
   310	  // Radius) EINMAL und benutzt den gerasterten Wert fuer Schluessel UND
   311	  // Zeichnung. Die alte Fassung hat genau das nicht getan.
   312	  //
   313	  //   mal()  zeichnet das Motiv um den URSPRUNG (0,0), Ausdehnung +-hw/+-hh
   314	  stempel(key, sig, x, y, hw, hh, mal) {
   315	    this._neuesBild();
   316	    if (!this.an) { this.stat.direkt++; this._umZeichnen(x, y, mal); return false; }
   317	    const c = ctx;
   318	    let t;
   319	    try { t = c.getTransform(); } catch (e) { this.stat.direkt++; this._umZeichnen(x, y, mal); return false; }
   320	    if (Math.abs(t.b) > 1e-6 || Math.abs(t.c) > 1e-6 || t.a <= 0 || t.d <= 0) {
   321	      this.stat.direkt++; this._umZeichnen(x, y, mal); return false;
   322	    }
   323	    const sx = t.a, sy = t.d;
   324	    const pw = Math.ceil(2 * hw * sx) + 2, ph = Math.ceil(2 * hh * sy) + 2;
   325	    // Ein Stempel lohnt nur, solange er KLEIN ist. Gespart wird der Aufbau
   326	    // des Motivs (ein Verlaufsobjekt, ein Weichzeichner-Durchlauf); die
   327	    // Flaeche, die gefuellt wird, kostet so oder so (Merksatz 10).
   328	    if (pw < 2 || ph < 2 || pw > this.MAX_STEMPEL || ph > this.MAX_STEMPEL) {
   329	      this.stat.direkt++; this._umZeichnen(x, y, mal); return false;
   330	    }
   331	    const s = (Array.isArray(sig) ? sig.join(",") : String(sig)) + "|" + pw + "x" + ph;
   332	    let e = this._m.get(key);
   333	    if (e && e.s === s) {
   334	      this._m.delete(key); this._m.set(key, e);
   335	      this.stat.treffer++;
   336	    } else {
   337	      if (this._budgetStempel <= 0) { this.stat.aufschub++; this.stat.direkt++; this._umZeichnen(x, y, mal); return false; }
   338	      this._budgetStempel--;
   339	      // Der Ursprung liegt in der Mitte des Blechs (plus ein Pixel Rand,
   340	      // damit ein weicher Rand nicht an der Kante abgeschnitten wird).
   341	      e = this._backe(key, e, s, pw, ph, 0, 0, sx, sy, pw / 2, ph / 2, mal);
   342	      if (!e) { this.stat.direkt++; this._umZeichnen(x, y, mal); return false; }
   343	    }
   344	    // 1:1 in der Groesse, frei in der Lage.
   345	    c.drawImage(e.cv, x - (pw / 2) / sx, y - (ph / 2) / sy, pw / sx, ph / sy);
   346	    return true;
   347	  },
   348	
   349	  _umZeichnen(x, y, mal) {
   350	    ctx.save(); ctx.translate(x, y); try { mal(); } finally { ctx.restore(); }
   351	  },
   352	
   353	  _backe(key, alt, s, pw, ph, x, y, sx, sy, fx, fy, mal) {
   354	    let e = alt;
   355	    try {
   356	      if (!e) {
   357	        e = { cv: document.createElement("canvas"), c2: null, s: "", bytes: 0 };
   358	        this._m.set(key, e);
   359	      } else {
   360	        this._m.delete(key); this._m.set(key, e);
   361	      }
   362	      // Eine Zuweisung an width/height setzt die Leinwand zurueck — das ist
   363	      // hier erwuenscht (leeres Blech) und spart ein clearRect.
   364	      e.cv.width = pw; e.cv.height = ph;
   365	      e.c2 = e.cv.getContext("2d");
   366	      if (!e.c2) throw new Error("kein 2D-Kontext");
   367	      this.bytes -= e.bytes;
   368	      e.bytes = pw * ph * 4;
   369	      this.bytes += e.bytes;
   370	      // Weltkoordinaten -> Blech: derselbe Massstab wie im Ziel, Ursprung
   371	      // auf die linke obere Ecke des Motivs geschoben. Damit zeichnet `mal`
   372	      // in exakt denselben Zahlen wie sonst auch. `fx`/`fy` setzen das Motiv
   373	      // dabei in genau die Unterpixel-Lage, in der es auf dem Bildschirm
   374	      // steht — sonst waeren alle Kanten um bis zu einen halben Pixel
   375	      // versetzt, und genau das sieht man an jeder Silhouette.
   376	      e.c2.setTransform(sx, 0, 0, sy, -x * sx + fx, -y * sy + fy);
   377	
   378	      // Der ganze Zeichencode des Projekts schreibt in den globalen `ctx`.
   379	      // Fuer die Dauer des Backens zeigt der auf das Blech — genau so, wie
   380	      // World.begin() ihn auf den Welt-Puffer umlegt.
   381	      const zurueck = ctx;
   382	      /* eslint-disable-next-line no-global-assign */
   383	      ctx = e.c2;
   384	      try { mal(); }
   385	      finally { /* eslint-disable-next-line no-global-assign */ ctx = zurueck; }
   386	
   387	      e.s = s;
   388	      // ===================== Die Bruecke zur GPU (#107, der Umbau) =========
   389	      // Im GPU-Weg (gpu.js) ist das Blech nicht mehr nur ein Sprite-Cache,
   390	      // sondern die Stelle, an der ein Motiv EINMAL zur Textur wird. Diese
   391	      // Zahl sagt der Senke, dass der Inhalt der Leinwand ein anderer ist -
   392	      // ohne sie zeigte die Textur fuer immer den ersten Backvorgang.
   393	      e.cv.__gpuVer = (e.cv.__gpuVer | 0) + 1;
   394	      this.stat.backen++;
   395	      this._raeumen();
   396	      return e;
   397	    } catch (err) {
   398	      console.warn("Blech: '" + key + "' liess sich nicht backen, wird direkt gezeichnet:", err);
   399	      this.vergiss(key);
   400	      return null;
   401	    }
   402	  },
   403	
   404	  _raeumen() {
   405	    if (this.bytes <= this.MAX_BYTES) return;
   406	    for (const k of this._m.keys()) {
   407	      if (this.bytes <= this.MAX_BYTES) break;
   408	      const e = this._m.get(k);
   409	      this.bytes -= e.bytes;
   410	      this._m.delete(k);
   411	      this._gpuWeg(e);
   412	      this.stat.raus++;
   413	    }
   414	  },
   415	
   416	  // Eine verdraengte Leinwand haelt im GPU-Weg noch eine Textur fest. Ohne
   417	  // dieses Aufraeumen wuechse der Texturspeicher genau um das, was der
   418	  // LRU-Deckel im Hauptspeicher gerade freigegeben hat.
   419	  _gpuWeg(e) {
   420	    if (e && e.cv && typeof Gpu !== "undefined" && Gpu.bereit) Gpu.vergiss(e.cv);
   421	  },
   422	
   423	  vergiss(key) {
   424	    const e = this._m.get(key);
   425	    if (!e) return;
   426	    this.bytes -= e.bytes;
   427	    this._m.delete(key);
   428	    this._gpuWeg(e);
   429	  },
   430	
   431	  // Alles wegwerfen. Bei einem Wechsel der Aufloesung nicht noetig (die
   432	  // Signatur traegt die Pixelgroesse), aber beim Ortswechsel spart es
   433	  // Speicher, den sonst erst die LRU findet.
   434	  leeren() {
   435	    for (const [, e] of this._m) this._gpuWeg(e);
   436	    this._m.clear();
   437	    this.bytes = 0;
   438	  },
   439	
   440	  // Fuer die Diagnose auf dem Geraet (perf.js) und fuer den Bericht.
   441	  bericht() {
   442	    const teile = [];
   443	    for (const [k, e] of this._m) teile.push({ k, kb: Math.round(e.bytes / 1024) });
   444	    teile.sort((a, b) => b.kb - a.kb);
   445	    return { an: this.an, mb: +(this.bytes / 1048576).toFixed(2), n: this._m.size,
   446	             jeBild: +this.jeBild(false).toFixed(3), stat: this.stat, teile };
   447	  }
   448	};

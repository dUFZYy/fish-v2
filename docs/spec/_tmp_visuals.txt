     3	// Sonnenstrahlen unter Wasser (God Rays) – von der Sonnenposition schräg nach unten, sanft schwankend
     4	function drawGodRays() {
     5	  const pal = getPalette(dayTime);
     6	  const light = pal.light;
     7	  const src = sunVisible && light > 0.35 ? sunX : (moonVisible ? moonX : null);
     8	  if (src == null) return;
     9	  const alpha = (sunVisible ? 0.10 * light : 0.05) * (getLocation().id === "tiefsee" ? 0.3 : 1);
    10	  ctx.save();
    11	  ctx.beginPath(); ctx.rect(0, horizonY, canvas.width, canvas.height - horizonY); ctx.clip();
    12	  ctx.globalCompositeOperation = "lighter";
    13	  for (let i = 0; i < 7; i++) {
    14	    const sway = Math.sin(time * 0.35 + i * 1.3) * 40;
    15	    const x0 = src + (i - 3) * 70 + sway;
    16	    const w = 26 + (i % 3) * 14;
    17	    const g = ctx.createLinearGradient(0, horizonY, 0, canvas.height);
    18	    g.addColorStop(0, `rgba(255,255,230,${alpha * (0.6 + 0.4 * Math.sin(time * 0.8 + i))})`);
    19	    g.addColorStop(1, "rgba(255,255,230,0)");
    20	    ctx.fillStyle = g;
    21	    ctx.beginPath();
    22	    ctx.moveTo(x0 - w, horizonY); ctx.lineTo(x0 + w, horizonY);
    23	    ctx.lineTo(x0 + w * 3 + 160, canvas.height); ctx.lineTo(x0 - w * 3 + 160, canvas.height);
    24	    ctx.closePath(); ctx.fill();
    25	  }
    26	  ctx.restore();
    27	}

    30	function drawCaustics() {
    31	  const light = getPalette(dayTime).light;
    32	  if (light < 0.3) return;
    33	  ctx.save();
    34	  ctx.globalCompositeOperation = "lighter";
    35	  ctx.strokeStyle = `rgba(255,255,255,${0.045 * light})`; ctx.lineWidth = 1.5;
    36	  for (let i = 0; i < 12; i++) {
    37	    const x = ((i * 137 + time * 18) % (canvas.width + 120)) - 60;
    38	    const y = horizonY + 25 + (i % 4) * 22 + Math.sin(time * 1.1 + i) * 6;
    39	    const r = 22 + (i % 3) * 10;
    40	    ctx.beginPath(); ctx.ellipse(x, y, r, r * 0.28, Math.sin(time * 0.5 + i) * 0.3, 0, Math.PI * 2); ctx.stroke();
    41	  }
    42	  ctx.restore();
    43	}

    45	// --- Spiegelung unter der Wasserlinie, in Canvas 2D (#107, Runde 3) ---------
    46	//
    47	// Warum das hier steht: seit Runde 3 startet ein Telefon in Canvas 2D, weil der
    48	// Hybrid dort nicht flüssig wurde. Damit fällt weg, was Dustin am Wasser am
    49	// meisten mag — Brechung, Spiegelung, Kaustik-Netz. Kaustik und Glitzern hat
    50	// der 2D-Weg noch (drawCaustics, drawWater); die SPIEGELUNG hatte er nie, und
    51	// sie ist die auffälligste der drei: ohne sie steht der Steg auf einer Fläche,
    52	// mit ihr steht er im Wasser.
    53	//
    54	// Der Kniff ist, dass diese Spiegelung fast nichts kostet. Sie kopiert einen
    55	// Streifen des Bildes ÜBER der Wasserlinie an sich selbst vorbei, gespiegelt,
    56	// nach unten — ein Blit, keine neue Zeichnung. Quelle und Ziel überlappen
    57	// nicht, deshalb braucht es keinen Zwischenpuffer (und genau ein solcher wäre
    58	// wieder die Vollbild-Kopie, die wir gerade losgeworden sind).
    59	//
    60	// Acht Streifen statt einem, weil ein einziger Blit eine harte Unterkante
    61	// hätte: so klingt die Spiegelung nach unten aus. Der seitliche Versatz je
    62	// Streifen folgt derselben Wellenformel wie die Oberfläche (getWave) — das ist
    63	// keine echte Brechung, aber es ist die Bewegung, die das Auge dort erwartet,
    64	// und sie kostet einen Sinus je Streifen.
    65	//
    66	// Ehrlich: das ist eine Annäherung. Der Shader spiegelt jedes Pixel einzeln
    67	// und krümmt es mit der Neigung der Welle; das hier spiegelt acht Bänder und
    68	// verschiebt sie seitlich. Aus einem Meter Abstand ist der Unterschied klein,
    69	// aus zehn Zentimetern nicht.
    70	//
    71	// Und sie ist das Erste, was fällt, wenn es eng wird. Gemessen (Chrome 151,
    72	// 1170×2532, CPU 6× gedrosselt, Median aus 200 Bildern) kostet sie 1,3 ms von
    73	// 11,2 — rund ein Zehntel des Bildes. Das ist wenig für das, was sie zeigt, und
    74	// trotzdem zu viel, wenn ein Gerät ohnehin schon über der Frist liegt: sie ist
    75	// Schmuck, und Schmuck darf nicht zwischen Dustin und einem spielbaren Bild
    76	// stehen. Ab 20 ms geglätteter Bildzeit bleibt sie weg, unter 17 ms kommt sie
    77	// zurück — die Lücke dazwischen ist die Hysterese, sonst blinkt sie.
    78	const MIRROR_SLICES = 8;
    79	let mirrorAus = false;
    80	function drawSurfaceMirror() {
    81	  if (typeof Perf !== "undefined" && Perf.g.takt > 1) {
    82	    if (Perf.g.takt > 20) mirrorAus = true;
    83	    else if (Perf.g.takt < 17) mirrorAus = false;
    84	    if (mirrorAus) return;
    85	  }
    86	  const loc = getLocation();
    87	  if (loc.mode === "dive" || loc.mode === "ice") return;       // kein offener Spiegel
    88	  const P = (typeof WATER_PROFILES !== "undefined" && WATER_PROFILES[loc.id]) || null;
    89	  const k = P ? P.mirror : 0.5;
    90	  if (!(k > 0.01)) return;
    91	  const bandH = Math.min(canvas.height * 0.13, 130);
    92	  if (bandH < 12 || horizonY < 20) return;
    93	  // Bildschirmpixel der Quelle: das Bild wurde unter DIESER Transformation
    94	  // gezeichnet (Wackeln, Hype-Zoom), also muss die Quelle durch dieselbe
    95	  // Abbildung. Sonst wandert die Spiegelung im großen Moment aus dem Bild.
    96	  const m = ctx.getTransform();
    97	  const sx = m.e, sw = m.a * canvas.width;
    98	  const sy = m.d * (horizonY - bandH) + m.f, sh = m.d * bandH;
    99	  if (sy < 0 || sh < 2 || sw < 2) return;   // Quelle unvollständig: lieber gar nicht
   100	  ctx.save();
   101	  ctx.beginPath(); ctx.rect(0, horizonY, canvas.width, bandH); ctx.clip();
   102	  // y -> 2*horizonY - y: der Streifen über der Linie landet unter ihr
   103	  ctx.translate(0, horizonY); ctx.scale(1, -1); ctx.translate(0, -horizonY);
   104	  const sliceH = bandH / MIRROR_SLICES, sliceSh = sh / MIRROR_SLICES;
   105	  for (let i = 0; i < MIRROR_SLICES; i++) {
   106	    // Achtung, die Reihenfolge dreht sich: Streifen 0 liegt am WEITESTEN über
   107	    // der Linie und landet gespiegelt am tiefsten. `t` ist deshalb die Tiefe
   108	    // des Ziels, nicht der Laufindex.
   109	    const t = (MIRROR_SLICES - i) / MIRROR_SLICES;     // 1 = tief unten, 1/N = an der Linie
   110	    const a = k * 0.5 * (1 - t) * (1 - t);             // klingt quadratisch aus
   111	    if (a < 0.012) continue;
   112	    // Versatz aus der Wellenformel: nah an der Linie ruhig, weiter unten mehr
   113	    const off = Math.sin(time * 1.9 + t * 5.4) * (1.5 + t * 5.0);
   114	    ctx.globalAlpha = a;
   115	    ctx.drawImage(canvasEl,
   116	      sx, sy + sliceSh * i, sw, sliceSh,
   117	      off, horizonY - bandH + sliceH * i, canvas.width, sliceH + 0.6);
   118	  }
   119	  ctx.restore();
   120	}

   123	function drawDepthFog() {
   124	  const g = ctx.createLinearGradient(0, horizonY, 0, canvas.height);
   125	  g.addColorStop(0, "rgba(0,10,30,0)"); g.addColorStop(0.55, "rgba(0,10,30,0.04)"); g.addColorStop(1, "rgba(0,8,25,0.22)");
   126	  ctx.fillStyle = g; ctx.fillRect(0, horizonY, canvas.width, canvas.height - horizonY);
   127	  const v = ctx.createRadialGradient(canvas.width / 2, canvas.height / 2, Math.min(canvas.width, canvas.height) * 0.45, canvas.width / 2, canvas.height / 2, Math.max(canvas.width, canvas.height) * 0.8);
   128	  v.addColorStop(0, "rgba(0,0,0,0)"); v.addColorStop(1, "rgba(0,0,20,0.28)");
   129	  ctx.fillStyle = v; ctx.fillRect(0, 0, canvas.width, canvas.height);
   130	}

   133	function drawFishShadows() {
   134	  ctx.save();
   135	  ctx.fillStyle = "rgba(0,10,30,0.18)";
   136	  for (const f of fishes) {
   137	    if (f.state === "fleeing" && f.alpha < 0.5) continue;

   140	  }
   141	  ctx.restore();

   144	// Ortswechsel: kurze Blende mit Ortsname
   145	let transition = null; // { t, loc }
   146	function startTransition(loc) { transition = { t: 0, loc }; }
   147	function updateTransition(dt) { if (!transition) return; transition.t += dt / 1.2; if (transition.t >= 1) transition = null; }
   148	function drawTransition() {
   149	  if (!transition) return;
   150	  const t = transition.t;
   151	  const a = t < 0.35 ? t / 0.35 : t > 0.65 ? (1 - t) / 0.35 : 1;
   152	  ctx.fillStyle = `rgba(4,12,28,${a * 0.96})`; ctx.fillRect(0, 0, canvas.width, canvas.height);
   153	  if (a > 0.6) {
   154	    const s = isNarrow() ? 1 : uiScale();
   155	    ctx.save(); ctx.globalAlpha = (a - 0.6) / 0.4; ctx.textAlign = "center"; ctx.textBaseline = "middle";
   156	    ctx.font = `${Math.round(56 * s)}px sans-serif`; ctx.fillText(transition.loc.icon, canvas.width / 2, canvas.height / 2 - 30 * s);
   157	    ctx.fillStyle = "#fff"; ctx.font = `bold ${Math.round(26 * s)}px 'Segoe UI', system-ui, sans-serif`;
   158	    ctx.fillText(transition.loc.name, canvas.width / 2, canvas.height / 2 + 30 * s);
   159	    ctx.fillStyle = "rgba(255,255,255,0.6)"; ctx.font = `${Math.round(13 * s)}px 'Segoe UI', system-ui, sans-serif`;
   160	    ctx.fillText(`${speciesForLocation(transition.loc.id).length} Arten · ${dexDiscoveredIn(transition.loc.id)} entdeckt`, canvas.width / 2, canvas.height / 2 + 58 * s);
   161	    ctx.restore();
   162	  }
